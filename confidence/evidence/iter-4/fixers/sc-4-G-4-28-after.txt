"""Run the run-steps of the sidecar job of a GitHub workflow file, in order, the way Actions runs
them (bash -e -o pipefail, working-directory sidecar, CI=true, stop at the first failing step).
Sandbox substitutions, printed when used: the kubeconform download step and the image build step
are skipped (github.com release downloads and the UBI registry are blocked; the tools are on PATH
and the image build is evidenced separately), and govulncheck gets -db pointing at the bucket
behind vuln.go.dev, which the proxy blocks.
usage: python3 -I ci_sidecar_steps.py <workflow.yml> <repo tree>"""
import os, subprocess, sys, yaml

workflow, tree = sys.argv[1], sys.argv[2]
job = yaml.safe_load(open(workflow))["jobs"]["sidecar"]
workdir = os.path.join(tree, job.get("defaults", {}).get("run", {}).get("working-directory", "."))
env = dict(os.environ, CI="true")
for i, step in enumerate(job["steps"]):
    if "run" not in step:
        continue
    name, cmd = step.get("name", ""), step["run"]
    if name in ("Install kubeconform", "Build the image"):
        print(f"--- step {i} {name!r}: skipped here (see header)")
        continue
    if "govulncheck@" in cmd:
        cmd = cmd.replace(" ./...", " -db https://storage.googleapis.com/go-vulndb ./...")
        print(f"--- step {i}: sandbox substitution: -db https://storage.googleapis.com/go-vulndb")
    print(f"--- step {i} {name!r}: {cmd.strip()}", flush=True)
    r = subprocess.run(["bash", "--noprofile", "--norc", "-eo", "pipefail", "-c", cmd], cwd=workdir, env=env,
                       stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    out = r.stdout.strip().splitlines()
    for line in (out if len(out) <= 12 else out[:3] + ["  [...]"] + out[-8:]):
        print("    " + line)
    print(f"    exit={r.returncode}", flush=True)
    if r.returncode != 0:
        print("JOB FAILED at this step; later steps do not run")
        sys.exit(1)
print("JOB PASSED (every step run here exited 0)")
